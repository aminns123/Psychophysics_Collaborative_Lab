from psychophysics_lab.researcher import AdaptiveAlgorithm, Step


class FixedStep:
    def __init__(self, *args, **kwargs):
        self.history = []

    def respond(self, identity, correct, update_value):
        # EDIT UPDATE RULE: experiment supplies the meaning/size of each step.
        step = Step.DOWN if correct else Step.UP
        value = update_value(step)
        self.history.append({"identity": identity, "correct": bool(correct), "value": value})
        return step, value

    def snapshot(self):
        return {"schema_version": 1, "history": list(self.history)}


SPEC = AdaptiveAlgorithm("BLUEPRINT_ID", "One-down/one-up fixed step", FixedStep)
